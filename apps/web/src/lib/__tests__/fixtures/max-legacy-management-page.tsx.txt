type Drink = {
  id: string; name: string; desc: string; price: number | null; priceLabel: string;
  category: string; color: string; foam: string; availability: string;
  image: string; options: string[];
};
type CartDrink = Omit<Drink, "price"> & { price: number; unitPriceKopecks: number; option?: string };
const ART: Record<string, { color: string; foam: string }> = {
  filter: { color: "#5B3A24", foam: "#5B3A24" },
  cappuccino: { color: "#8A5A3A", foam: "#F3E8D4" },
  latte: { color: "#B8895F", foam: "#F7EFE0" },
  cocoa: { color: "#6B3E2E", foam: "#E9D6C3" },
};
function toDrink(item: YleumMaxContentItem): Drink {
  return { id: item.id, name: item.title, desc: item.description,
    price: parseRubPrice(item.price), priceLabel: item.price,
    category: item.category, availability: item.availability,
    image: item.image_url, options: item.options,
    ...(ART[item.id] ?? { color: "#5B3A24", foam: "#F3E8D4" }) };
}
function safeTrack(type: Parameters<typeof trackMaxActivity>[0], eventId?: string) {
  try { void trackMaxActivity(type, eventId ? { eventId } : undefined).catch(() => undefined); } catch { /* Telemetry never blocks ordering. */ }
}

type OrderItem = { id: string; name: string; price: number; qty: number; option?: string };
type Order = {
  id: string;
  customerName: string;
  items: OrderItem[];
  total: number;
  status: string;
  createdAt: string;
  pickupOption?: string | null;
  pickupAt?: string | null;
  baristaComment?: string | null;
};
const STATUS_LABELS: Record<string, string> = {
  new: "Новый", in_progress: "В работе", completed: "Выполнен", cancelled: "Отменён",
};
const statusLabel = (status: string) => STATUS_LABELS[status] ?? status;
function sumOrderRub(records: readonly ManagedRecord[]): number {
  let total = 0;
  for (const record of records) total += (record.total_kopecks ?? 0) / 100;
  return total;
}
function mapManagedRecord(record: ManagedRecord): Order {
  if (record.resource !== "orders" || record.total_kopecks == null || record.currency !== "RUB") throw new Error("Invalid order receipt");
  const items: OrderItem[] = [];
  for (const item of record.items) items.push({ id: item.catalog_id, name: item.title,
    price: item.unit_price_kopecks / 100, qty: item.quantity, option: item.option });
  const pickup = record.fulfillment;
  const pickupIds: Record<number, string> = { 0: "asap", 15: "m15", 30: "m30", 60: "h1" };
  return { id: record.id, customerName: record.customer_name,
    items,
    total: record.total_kopecks / 100, status: record.status, createdAt: record.created_at,
    pickupOption: pickup ? pickupIds[pickup.delay_minutes] ?? null : null,
    pickupAt: pickup?.scheduled_for ?? null, baristaComment: record.comment ?? null };
}
function businessError(error: unknown, fallback: string): string {
  if (error instanceof ManagedBusinessError && (error.status === 401 || error.status === 403)) return "Нет доступа. Откройте приложение из MAX.";
  return fallback;
}

type PickupOption = "asap" | "m15" | "m30" | "h1";

  { id: "m30", label: "Через 30 минут", minutes: 30 },
  { id: "h1", label: "Через час", minutes: 60 },
];
const COMMENT_MAX = 280;
const fmtTime = (d: Date) => d.toLocaleTimeString("ru-RU", { hour: "2-digit", minute: "2-digit" });
const optionLabel = (id: string | null | undefined) => PICKUP_OPTIONS.find((p) => p.id === id)?.label ?? null;
const leftText = (ms: number) => {
  const m = Math.max(0, Math.ceil(ms / 60000));
  if (m < 1) return "меньше минуты";
  if (m < 60) return `${m} мин`;
  const h = Math.floor(m / 60);
  const r = m % 60;
  return r ? `${h} ч ${r} мин` : `${h} ч`;
};

const pickupSummary = (o: { pickupAt?: string | null; pickupOption?: string | null }, now: number) => {
  if (!o.pickupAt) return "Обычный самовывоз";
  const at = new Date(o.pickupAt);
  if (Number.isNaN(at.getTime())) return "Обычный самовывоз";
  const diff = at.getTime() - now;
  const base = `Забрать к ${fmtTime(at)}`;
  return diff > 0 ? `${base} · осталось ${leftText(diff)}` : `${base} · время получения наступило`;
};

        <p className="sever-sub">{appProfile.profile?.summary}</p>
        {appProfile.status === "error" && <div className="sever-err" role="alert">
          Не удалось загрузить данные приложения. <button className="sever-btn" onClick={() => void appProfile.refresh()}>Повторить загрузку данных</button>
        </div>}

        {!recoveryReady && <div className="sever-note" role={recoveryState === "error" ? "alert" : "status"}>
          {recoveryState === "error" ? recoveryError : "Проверяем отправки перед новым заказом или заявкой…"}
          {recoveryState === "error" && <p><button className="sever-btn" onClick={() => void restorePending()}>Повторить проверку отправок</button></p>}
        </div>}

        {tab === "menu" && (
          <section aria-label="Меню">
            <h2 className="sever-h2">Меню напитков</h2>
            <div className="sever-chips" role="group" aria-label="Фильтр меню">
              {[[null, "Все"], ...categories.map(category => [category, category])].map(([k, l]) => (
                <button key={k === null ? "all" : "category:" + k} className="sever-chip" aria-pressed={selection.category === k} onClick={() => selection.setCategory(k)}>
                  {l}
                </button>
              ))}
            </div>
            {catalog.status === "loading" && <div className="sever-empty" role="status">Загружаем меню…</div>}
            {catalog.status === "error" && <div className="sever-err" role="alert">
              Не удалось загрузить меню. <button className="sever-btn" onClick={() => void catalog.refresh()}>Повторить</button>
            </div>}
            {catalog.status === "ready" && visible.length === 0 && <div className="sever-empty">В этом разделе пока нет позиций.</div>}
            <div className="sever-list">
              {catalog.items.map(item => {
                if (selection.category !== null && item.category !== selection.category) return null;
                const d = toDrink(item);
                const q = cart[d.id] ?? 0;
                return (
                  <div className="sever-item" key={d.id} onClick={() => safeTrack("item_view")}>
                    <div className="sever-art">{d.image ? (
                      // Owner catalog images may use arbitrary public hosts.
                      // eslint-disable-next-line @next/next/no-img-element
                      <img src={d.image} alt="" width={48} height={48} style={{ objectFit: "cover", borderRadius: 8 }} />
                    ) : <Cup color={d.color} foam={d.foam} />}</div>
                    <div className="sever-info">
                      <h2 className="sever-name">{d.name}</h2>
                      <p className="sever-desc">{d.desc}</p>
                      <span className="sever-price">{d.price === null ? d.priceLabel : rub(d.price)}</span>
                      {d.availability !== "in_stock" && <p className="sever-muted">{d.availability === "on_request" ? "Под заказ" : "Нет в наличии"}</p>}
                      {d.price === null && <p className="sever-muted">Уточните цену в кофейне</p>}
                      {d.options.length > 0 && <select className="sever-input" aria-label={`Вариант: ${d.name}`}
                        disabled={q > 0 || submitting || orderUnknown || !recoveryReady} value={selection.getOption(d) ?? ""}
                        onChange={event => selection.selectOption(d.id, event.target.value)}>
                        <option value="">Выберите вариант</option>
                        {d.options.map(option => <option key={option} value={option}>{option}</option>)}
                      </select>}
                    </div>
                    {q === 0 ? (
                      <button className="sever-btn sever-add" aria-label={`Добавить в корзину: ${d.name}`}
                        disabled={submitting || orderUnknown || !recoveryReady || d.price === null || d.availability !== "in_stock" ||
                          (d.options.length > 0 && selection.getOption(d) === undefined)}
                        onClick={() => setQty(d.id, 1)}>
                        <Plus size={20} aria-hidden />
                      </button>
                    ) : (
                      <div className="sever-qty">
                        <button aria-label={`Уменьшить количество: ${d.name}`} onClick={() => setQty(d.id, q - 1)}><Minus size={18} aria-hidden /></button>
                        <output aria-label={`Количество ${d.name}`}>{q}</output>
                        <button aria-label={`Увеличить количество: ${d.name}`} onClick={() => setQty(d.id, q + 1)}><Plus size={18} aria-hidden /></button>
                      </div>
                    )}
                  </div>
                );
              })}
            </div>
            {count > 0 && (
              <p style={{ marginTop: 16 }}>
                <button className="sever-btn block" onClick={() => setTab("cart")}>
                  Корзина: {count} · {rub(total)}
                </button>
              </p>
            )}
          </section>
        )}

        {tab === "cart" && (
          <section aria-label="Корзина">
            <h2 className="sever-h2">Корзина</h2>
            {submitError && <div className="sever-err" role="alert">{submitError}</div>}
            {lastOrder && lines.length === 0 && (
              <div className="sever-ok" role="status">
                <Check size={18} aria-hidden style={{ verticalAlign: "-3px" }} /> Заказ сохранён, {lastOrder.customerName}! Статус: {statusLabel(lastOrder.status)}. Сумма {rub(lastOrder.total)}. Заберите его в кофейне — оплата не нужна.
                <br /><strong data-testid="confirm-pickup">{pickupSummary(lastOrder, now)}</strong>
                {lastOrder.baristaComment ? (<><br /><span data-testid="confirm-comment">Комментарий бариста: {lastOrder.baristaComment}</span></>) : null}
                <p style={{ margin: "10px 0 0" }}>
                  <button className="sever-btn ghost" onClick={() => setTab("orders")}>Открыть «История заказов»</button>
                </p>
              </div>
            )}
            {orderUnknown && orderAttempt.current && <div className="sever-card" aria-label="Проверка отправки заказа">
              <p className="sever-note" role="status">Есть неподтверждённая отправка заказа. Данные зафиксированы; проверка не создаёт дубликат.</p>
              <p>Позиций: {orderAttempt.current.items.length} · Сумма отправки: {rub(orderAttempt.current.items.reduce((sum, item) => sum + item.unitPriceKopecks * item.quantity, 0) / 100)}</p>
              <label className="sever-field" htmlFor="sever-name"><span>Имя в отправленном заказе</span>
                <input id="sever-name" className="sever-input" value={name} disabled /></label>
              <label className="sever-field" htmlFor="sever-comment"><span>Сохранённый комментарий бариста</span>
                <textarea id="sever-comment" className="sever-input" value={baristaComment} rows={3} disabled /></label>
              <p className="sever-muted">{orderAttempt.current.fulfillment
                ? "Самовывоз: " + (orderAttempt.current.fulfillment.delayMinutes === 0 ? "как можно скорее" : "через " + orderAttempt.current.fulfillment.delayMinutes + " мин") + ". Время получения подтвердит сервер."
                : "Обычный самовывоз. Время получения не указано."}</p>
              <button className="sever-btn block" disabled={submitting || !recoveryReady} onClick={submit}>
                {submitting ? "Проверяем…" : "Проверить отправку"}
              </button>
            </div>}
            {lines.length === 0 && !orderUnknown && (
              !lastOrder && (
                <div className="sever-empty">
                  Корзина пуста. Выберите напиток в меню.
                  <p style={{ margin: "12px 0 0" }}>
                    <button className="sever-btn" onClick={() => setTab("menu")}>Перейти в меню</button>
                  </p>
                </div>
              )
            )}
            {lines.length > 0 && !orderUnknown && (
              <>
                <div className="sever-list">
                  {lines.map((l) => (
                    <div className="sever-item" key={l.id}>
                      <div className="sever-info">
                        <h3 className="sever-name" style={{ fontSize: 17 }}>{l.name}</h3>
                        {l.option && <p className="sever-muted">{l.option}</p>}
                        <span className="sever-muted">{rub(l.price)} × {l.qty} = </span>
                        <span className="sever-price">{rub(l.price * l.qty)}</span>
                      </div>
                      <div className="sever-qty">
                        <button aria-label={`Уменьшить количество: ${l.name}`} onClick={() => setQty(l.id, l.qty - 1)}><Minus size={18} aria-hidden /></button>
                        <output aria-label={`Количество ${l.name}`}>{l.qty}</output>
                        <button aria-label={`Увеличить количество: ${l.name}`} onClick={() => setQty(l.id, l.qty + 1)}><Plus size={18} aria-hidden /></button>
                      </div>
                      <button className="sever-del" aria-label={`Удалить из корзины: ${l.name}`} onClick={() => setQty(l.id, 0)}>
                        <Trash2 size={20} aria-hidden />
                      </button>
                    </div>
                  ))}
                </div>
                <div className="sever-total"><span>Итого</span><strong>{rub(total)}</strong></div>
                <label className="sever-field" htmlFor="sever-name">
                  <span>Ваше имя</span>
                  <input
                    id="sever-name"
                    disabled={submitting || orderUnknown || !recoveryReady}
                    className="sever-input"
                    value={name}
                    maxLength={60}
                    autoComplete="given-name"
                    placeholder="Например, Анна"
                    onChange={(e) => setName(e.target.value)}
                  />
                </label>
                <fieldset className="sever-field" style={{ border: 0, padding: 0, margin: "12px 0 0" }}>
                  <legend style={{ fontWeight: 700, marginBottom: 8 }}>Время самовывоза</legend>
                  <div role="radiogroup" aria-label="Время самовывоза" style={{ display: "grid", gridTemplateColumns: "1fr 1fr", gap: 8 }}>
                    {PICKUP_OPTIONS.map((p) => {
                      const active = p.id === pickupOption;
                      return (
                        <button
                          key={p.id}
                          type="button"
                          role="radio"
                          aria-checked={active}
                          data-testid={`pickup-${p.id}`}
                          className={active ? "sever-btn" : "sever-btn ghost"}
                          style={{ minHeight: 44 }}
                          disabled={submitting || orderUnknown || !recoveryReady}
                          onClick={() => setPickupOption(p.id)}
                        >
                          {p.label}
                        </button>
                      );
                    })}
                  </div>
                </fieldset>
                <div className="sever-note" data-testid="pickup-preview" role="status">
                  <strong>Забрать к {fmtTime(pickupPreview)}</strong>
                  <br />
                  {selectedPickup.label} · {count} поз. на {rub(total)}
                </div>
                <label className="sever-field" htmlFor="sever-comment">
                  <span>Комментарий бариста (необязательно)</span>
                  <textarea
                    id="sever-comment"
                    disabled={submitting || orderUnknown || !recoveryReady}
                    className="sever-input"
                    rows={3}
                    value={baristaComment}
                    aria-invalid={commentTooLong}
                    aria-describedby="sever-comment-count"
                    placeholder="Например, без сахара"
                    onChange={(e) => setBaristaComment(e.target.value)}
                  />
                  <span id="sever-comment-count" className="sever-muted" data-testid="comment-count">
                    {commentTrimmed.length} / {COMMENT_MAX}
                  </span>
                </label>
                {commentTooLong && (
                  <div className="sever-err" role="alert">
                    Комментарий длиннее {COMMENT_MAX} символов на {commentTrimmed.length - COMMENT_MAX}. Сократите текст — мы не обрезаем его молча.
                  </div>
                )}
                <div className="sever-note">Тестовый заказ на самовывоз. Оплата не требуется, деньги не списываются.</div>
                <button className="sever-btn block" disabled={submitting || !recoveryReady} onClick={submit}>
                  {submitting ? <Loader2 size={20} className="sever-spin" aria-hidden /> : <Check size={20} aria-hidden />}
                  {submitting ? "Оформляем…" : orderUnknown ? "Повторить отправку заказа" : needsConfirmation ? "Подтвердить новую сумму" : "Подтвердить заказ"}
                </button>
              </>
            )}
          </section>
        )}

        {tab === "orders" && (
          <section aria-label="История заказов">
            <h2 className="sever-h2">История заказов</h2>
            {(ordersState === "loading" || ordersState === "idle") && (
              <div className="sever-empty" role="status"><Loader2 size={22} className="sever-spin" aria-hidden /><br />Загружаем заказы…</div>
            )}
            {ordersState === "error" && <div className="sever-err" role="alert">
              {ordersError}<p style={{ margin: "10px 0 0" }}><button className="sever-btn" onClick={() => void loadOrders()}>Повторить</button></p>
            </div>}
            {ordersState === "success" && orders.length === 0 && <div className="sever-empty">
              Пока нет заказов. Оформите первый — он появится здесь.
              <p style={{ margin: "12px 0 0" }}><button className="sever-btn" onClick={() => setTab("menu")}>Перейти в меню</button></p>
            </div>}
            {ordersState === "success" && orders.map(record => { const o = mapManagedRecord(record); return (
                <article className="sever-card" key={o.id}>
                  <div className="sever-row">
                    <strong>{record.customer_name}</strong>
                    <span className="sever-badge">{statusLabel(o.status)}</span>
                  </div>
                  <p className="sever-muted" style={{ margin: "4px 0 8px" }}>
                    {fmtDate(o.createdAt)}
                    <br /><span data-testid="orders-tab-pickup">{pickupSummary(o, now)}</span>
                    {o.baristaComment ? (
                      <>
                        <br /><span data-testid="orders-tab-comment">Комментарий бариста: {o.baristaComment}</span>
                      </>
                    ) : null}
                  </p>
                  <ul className="sever-tl">
                    {o.items.map((it) => (
                      <li key={it.id}><span style={{ flex: 1 }}>{it.name}{it.option ? ` (${it.option})` : ""} × {it.qty}</span><span>{rub(it.price * it.qty)}</span></li>
                    ))}
                  </ul>
                  <div className="sever-row" style={{ marginTop: 8 }}><span className="sever-muted">Итого, самовывоз</span><strong>{rub(o.total)}</strong></div>
                </article>
            ); })}
          </section>
        )}

        {tab === "profile" && (
          <section aria-label="Профиль">
            <h2 className="sever-h2">Профиль</h2>
            <div className="sever-card">
              {appProfile.status === "loading" && <p role="status">Загружаем данные приложения…</p>}
              {appProfile.profile && <>
                <strong style={{ fontSize: 18 }}>{appProfile.profile.app_name}</strong>
                <p className="sever-muted">{appProfile.profile.summary}</p>
                {appProfile.profile.support.email && <p><a href={"mailto:" + appProfile.profile.support.email}>{appProfile.profile.support.email}</a></p>}
                <p className="sever-muted">{appProfile.profile.support.response_time}</p>
              </>}
              <p className="sever-muted">Заказов: {orders.length} · Сумма заказов: {rub(sumOrderRub(orders))}</p>
            </div>
            <h2 className="sever-h2" style={{ marginTop: 20 }}>Заявка в кофейню</h2>
            {leadUnknown && <p className="sever-note" role="status">Есть неподтверждённая отправка заявки. Сообщение зафиксировано; сначала проверьте результат.</p>}
            <form className="sever-card" onSubmit={submitLead}>
              <label className="sever-field" htmlFor="sever-lead-name"><span>Ваше имя для заявки</span>
                <input id="sever-lead-name" className="sever-input" value={leadName} maxLength={60}
                  disabled={leadPending || leadUnknown || !recoveryReady} autoComplete="given-name"
                  onChange={event => setLeadName(event.target.value)} /></label>
              <label className="sever-field" htmlFor="sever-lead-comment"><span>Сообщение</span>
                <textarea id="sever-lead-comment" className="sever-input" rows={3} value={leadComment}
                  disabled={leadPending || leadUnknown || !recoveryReady} aria-invalid={leadComment.trim().length > 2000}
                  onChange={event => setLeadComment(event.target.value)} /></label>
              <p className="sever-muted">{leadComment.trim().length} / 2000</p>
              {leadError && <p className="sever-err" role="alert">{leadError}</p>}
              {leadSaved && <p className="sever-ok" role="status">Заявка сохранена. Её статус появится в истории ниже.</p>}
              <button className="sever-btn block" type="submit" disabled={leadPending || !recoveryReady}>
                {leadPending ? "Отправляем…" : leadUnknown ? "Проверить отправку" : "Отправить заявку"}
              </button>
            </form>
            <h2 className="sever-h2" style={{ marginTop: 20 }}>История заявок</h2>
            {(leadsState === "idle" || leadsState === "loading") && <p className="sever-empty" role="status">Загружаем заявки…</p>}
            {leadsState === "error" && <div className="sever-err" role="alert">{leadsError}
              <p><button className="sever-btn" onClick={() => void loadLeads()}>Повторить загрузку заявок</button></p>
            </div>}
            {leadsState === "success" && leads.length === 0 && <p className="sever-empty">Заявок пока нет.</p>}
            {leads.map(record => <article className="sever-card" key={record.id}>
              <div className="sever-row"><strong>{record.customer_name}</strong><span className="sever-badge">{statusLabel(record.status)}</span></div>
              <p className="sever-muted">{fmtDate(record.created_at)}</p><p>{record.comment}</p>
            </article>)}
            <h2 className="sever-h2" style={{ marginTop: 20 }}>История действий</h2>
            {historyBlock}
          </section>
        )}
      </div>

      <nav className="sever-nav" aria-label="Основная навигация">
        <div className="sever-nav-in">
          {tabs.map((t) => (
            <button key={t.id} className="sever-tab" aria-current={tab === t.id ? "page" : undefined} onClick={() => setTab(t.id)}>
              {t.icon}
              <span>{t.label}</span>
              {t.badge ? <i aria-label={`Позиций: ${t.badge}`}>{t.badge}</i> : null}
            </button>
          ))}
        </div>
      </nav>
    </main>
  );
}
